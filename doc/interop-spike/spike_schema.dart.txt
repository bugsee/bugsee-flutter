import 'package:pigeon/pigeon.dart';

@ConfigurePigeon(
  PigeonOptions(
    appDirectory: './',
    dartOut: 'lib/src/spike.g.dart',
    dartOptions: DartOptions(),
    kotlinOut: 'android/app/src/main/kotlin/com/bugsee/interop_spike/Spike.g.kt',
    kotlinOptions: KotlinOptions(package: 'com.bugsee.interop_spike', useJni: true),
    swiftOut: 'ios/Runner/Spike.g.swift',
    swiftOptions: SwiftOptions(useFfi: true, ffiModuleName: 'Runner'),
  ),
)
@HostApi()
abstract class SpikeApi {
  void ping();
  int echo(int value);
  void publishInts(Int32List rects);
  void publishList(List<int> rects);
  String thread();
}

class Evt {
  Evt({required this.name, this.data});
  String name;
  Map<String, String>? data;
}

@FlutterApi()
abstract class SpikeEvents {
  void onEvent(Evt event);
}

@HostApi()
abstract class SpikeTrigger {
  void fire(String name);
}
